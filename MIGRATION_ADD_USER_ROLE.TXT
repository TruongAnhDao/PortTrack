USE port_track;

-- Run this once for an existing PortTrack database.
-- Existing room owners become lecturers. All other accounts become students.
ALTER TABLE users
    ADD COLUMN role ENUM('LECTURER', 'STUDENT') NULL
        AFTER password_hash;

UPDATE users
SET role = CASE
    WHEN EXISTS (
        SELECT 1
        FROM rooms
        WHERE rooms.owner_id = users.id
    ) THEN 'LECTURER'
    ELSE 'STUDENT'
END;

ALTER TABLE users
    MODIFY COLUMN role ENUM('LECTURER', 'STUDENT') NOT NULL;
